C Program (Linear Probing Hashing)

#include 
#define SIZE 10

int hashTable[SIZE];

void initialize() {
    for (int i = 0; i < SIZE; i++)
        hashTable[i] = -1;
}

void insert(int value) {
    int index = value % SIZE;

    for (int i = 0; i < SIZE; i++) {
        int probeIndex = (index + i) % SIZE;
        if (hashTable[probeIndex] == -1) {
            hashTable[probeIndex] = value;
            printf("Inserted %d at index %d\\n", value, probeIndex);
            return;
        }
    }

    printf("Hash table is full! Cannot insert %d\\n", value);
}

void display() {
    printf("\\nHash Table:\\n");
    for (int i = 0; i < SIZE; i++) {
        if (hashTable[i] == -1)
            printf("[%d]  -\\n", i);
        else
            printf("[%d]  %d\\n", i, hashTable[i]);
    }
}

int main() {
    int n, value;

    initialize();

    printf("Enter number of values to insert: ");
    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        printf("Enter value %d: ", i + 1);
        scanf("%d", &value);
        insert(value);
    }

    display();

    return 0;
}
